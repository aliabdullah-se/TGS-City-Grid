#pragma once
#include <string>

struct Incident {
    int id = 0;
    int junctionId = 0;
    std::string category;
    int severity = 0;
    int timestamp = 0;
    std::string status = "REPORTED";
};

inline bool higherPriority(const Incident* a, const Incident* b) {
    if (a->severity != b->severity)
        return a->severity > b->severity;

    if (a->timestamp != b->timestamp)
        return a->timestamp < b->timestamp;

    return a->id < b->id;
}

std::string CityGrid::trim(const std::string& text) {
    std::string::size_type first =
        text.find_first_not_of(" \t");

    if (first == std::string::npos)
        return "";

    std::string::size_type last =
        text.find_last_not_of(" \t");

    return text.substr(first, last - first + 1);
}

bool CityGrid::blank(const std::string& text) {
    return trim(text).empty();
}

bool CityGrid::fields(const std::string& line, int needed,
                      std::string result[]) {
    std::string::size_type start = 0;
    int count = 0;

    for (std::string::size_type i = 0;
         i <= line.size(); ++i) {

        if (i == line.size() || line[i] == '|') {
            if (count >= needed)
                return false;

            result[count++] =
                trim(line.substr(start, i - start));

            start = i + 1;
        }
    }

    return count == needed;
}

bool CityGrid::number(const std::string& text,
                      int minimum, int maximum,
                      int& result) {
    if (text.empty() || text.size() > 9)
        return false;

    int value = 0;

    for (std::string::size_type i = 0;
         i < text.size(); ++i) {

        if (text[i] < '0' || text[i] > '9')
            return false;

        value = value * 10 + (text[i] - '0');
    }

    if (value < minimum || value > maximum)
        return false;

    result = value;
    return true;
}

void CityGrid::loadJunctions(
    std::ifstream& input,
    const std::string& filename) {

    std::string line;
    int lineNumber = 0;
    int accepted = 0;
    int rejected = 0;

    while (std::getline(input, line)) {
        ++lineNumber;

        // Remove Windows carriage return.
        if (!line.empty() &&
            line[line.size() - 1] == '\r')
            line.erase(line.size() - 1);

        if (blank(line))
            continue;

        std::string part[3];
        int id = 0;
        std::string reason;

        if (!fields(line, 3, part))
            reason = "wrong field count";
        else if (!number(part[0], 1, 999999999, id))
            reason = "invalid junction ID";
        else if (part[1].empty())
            reason = "empty name";
        else if (!junctionType(part[2]))
            reason = "invalid type";
        else if (junctions.find(id) != nullptr)
            reason = "duplicate junction ID";

        if (!reason.empty()) {
            std::cout << "ERROR " << filename
                      << ':' << lineNumber
                      << ": " << reason << '\n';

            ++rejected;
            continue;
        }

        Junction junction;
        junction.id = id;
        junction.name = part[1];
        junction.type = part[2];

        junctions.insert(id, junction);
        ++accepted;
    }

    std::cout << "LOAD " << filename << ": "
              << accepted << " accepted, "
              << rejected << " rejected\n";
}

void CityGrid::loadIncidents(
    std::ifstream& input,
    const std::string& filename) {

    std::string line;
    int lineNumber = 0;
    int accepted = 0;
    int rejected = 0;

    while (std::getline(input, line)) {
        ++lineNumber;

        if (!line.empty() &&
            line[line.size() - 1] == '\r')
            line.erase(line.size() - 1);

        if (blank(line))
            continue;

        std::string part[5];
        int id = 0;
        int junctionId = 0;
        int severity = 0;
        int timestamp = 0;
        std::string reason;

        if (!fields(line, 5, part))
            reason = "wrong field count";
        else if (!number(part[0], 1, 999999999, id))
            reason = "invalid incident ID";
        else if (!number(part[1], 1, 999999999,
                         junctionId))
            reason = "invalid junction ID";
        else if (!incidentCategory(part[2]))
            reason = "invalid category";
        else if (!number(part[3], 1, 5, severity))
            reason = "invalid severity";
        else if (!number(part[4], 0, 999999999,
                         timestamp))
            reason = "invalid timestamp";
        else if (incidentIndex.find(id) != nullptr)
            reason = "duplicate incident ID";
        else if (junctions.find(junctionId) == nullptr)
            reason = "junction " +
                     std::to_string(junctionId) +
                     " not found";

        if (!reason.empty()) {
            std::cout << "ERROR " << filename
                      << ':' << lineNumber
                      << ": " << reason << '\n';

            ++rejected;
            continue;
        }

        Incident* incident = new Incident;

        incident->id = id;
        incident->junctionId = junctionId;
        incident->category = part[2];
        incident->severity = severity;
        incident->timestamp = timestamp;
        incident->status = "REPORTED";

        // BST owns the Incident object.
        incidents.insert(id, incident);

        // HashMap stores the same pointer.
        incidentIndex.insert(id, incident);

        ++accepted;
    }

    std::cout << "LOAD " << filename << ": "
              << accepted << " accepted, "
              << rejected << " rejected\n";
}

bool CityGrid::run(
    const std::string& junctionPath,
    const std::string& incidentPath,
    const std::string& scriptPath) {

    std::ifstream junctionFile(junctionPath);

    if (!junctionFile) {
        std::cout << "FATAL: cannot open "
                  << junctionPath << '\n';
        return false;
    }

    std::ifstream incidentFile(incidentPath);

    if (!incidentFile) {
        std::cout << "FATAL: cannot open "
                  << incidentPath << '\n';
        return false;
    }

    std::ifstream scriptFile(scriptPath);

    if (!scriptFile) {
        std::cout << "FATAL: cannot open "
                  << scriptPath << '\n';
        return false;
    }

    loadJunctions(junctionFile, junctionPath);
    loadIncidents(incidentFile, incidentPath);

    executeScript(scriptFile);
    return true;
}

void CityGrid::printJunction(const Junction& junction) {
    std::cout << "JUNCTION " << junction.id << ' '
              << junction.name << ' ' << junction.type << '\n';
}

void CityGrid::printIncident(const Incident& incident) {
    std::cout << "INCIDENT " << incident.id
              << " junction=" << incident.junctionId
              << " category=" << incident.category
              << " severity=" << incident.severity
              << " time=" << incident.timestamp
              << " status=" << incident.status << '\n';
}

void CityGrid::closeIncident(int id) {
    Incident* incident = incidentIndex.find(id);

    if (incident->status == "PENDING")
        pending.removeById(id);

    // Remove references before the BST frees the object.
    incidentIndex.remove(id);
    incidents.remove(id);
}

void CityGrid::executeScript(std::ifstream& script) {
    std::string line;
    int lineNumber = 0;

    while (std::getline(script, line)) {
        ++lineNumber;

        if (!line.empty() && line[line.size() - 1] == '\r')
            line.erase(line.size() - 1);

        line = trim(line);

        if (line.empty() || line[0] == '#')
            continue;

        std::cout << "> " << line << '\n';
        executeCommand(line, lineNumber);
    }
}

void CityGrid::executeCommand(
    const std::string& line, int lineNumber) {

    std::istringstream input(line);
    std::string word[6];
    std::string token;
    int wordCount = 0;

    while (input >> token) {
        if (wordCount < 6)
            word[wordCount] = token;
        ++wordCount;
    }

    const std::string& command = word[0];
    int expected = -1;

    if (command == "CITY_ADD")
        expected = 3;
    else if (command == "CITY_FIND")
        expected = 1;
    else if (command == "CITY_LIST" ||
             command == "CITY_PREORDER")
        expected = 0;
    else if (command == "INC_REPORT")
        expected = 5;
    else if (command == "INC_FIND" ||
             command == "INC_CLOSE" ||
             command == "REQUEST")
        expected = 1;
    else if (command == "INC_LIST" ||
             command == "INC_PREORDER" ||
             command == "PRIORITY_NEXT" ||
             command == "PRIORITY_SERVE" ||
             command == "PRIORITY_COUNT" ||
             command == "HASH_STATS")
        expected = 0;

    std::string reason;

    if (expected < 0)
        reason = "unknown command " + command;
    else if (wordCount != expected + 1)
        reason = "wrong argument count";

    if (!reason.empty()) {
        std::cout << "ERROR line " << lineNumber
                  << ": " << reason << '\n';
        return;
    }

    int id = 0;
    int junctionId = 0;
    int severity = 0;
    int timestamp = 0;

    if (command == "CITY_ADD") {

        if (!number(word[1], 1, 999999999, id))
            reason = "invalid junction ID";
        else if (word[2].find('|') != std::string::npos)
            reason = "invalid name";
        else if (!junctionType(word[3]))
            reason = "invalid type";
        else if (junctions.find(id) != nullptr)
            reason = "duplicate junction ID";

        if (reason.empty()) {
            Junction junction;
            junction.id = id;
            junction.name = word[2];
            junction.type = word[3];

            junctions.insert(id, junction);
            std::cout << "JUNCTION ADDED " << id << '\n';
        }

    } else if (command == "CITY_FIND") {

        if (!number(word[1], 1, 999999999, id)) {
            reason = "invalid junction ID";
        } else {
            const Junction* junction = junctions.find(id);

            if (junction == nullptr)
                reason = "junction " + std::to_string(id)
                         + " not found";
            else
                printJunction(*junction);
        }

    } else if (command == "CITY_LIST") {

        std::cout << "JUNCTIONS " << junctions.size() << '\n';

        junctions.inOrder(
            [](const int&, const Junction& junction) {
                printJunction(junction);
            }
        );

    } else if (command == "CITY_PREORDER") {

        std::cout << "JUNCTION PREORDER";

        if (junctions.size() == 0) {
            std::cout << " EMPTY";
        } else {
            junctions.preOrder(
                [](const int& key, const Junction&) {
                    std::cout << ' ' << key;
                }
            );
        }

        std::cout << '\n';

    } else if (command == "INC_REPORT") {

        if (!number(word[1], 1, 999999999, id))
            reason = "invalid incident ID";
        else if (!number(word[2], 1, 999999999,
                         junctionId))
            reason = "invalid junction ID";
        else if (!incidentCategory(word[3]))
            reason = "invalid category";
        else if (!number(word[4], 1, 5, severity))
            reason = "invalid severity";
        else if (!number(word[5], 0, 999999999,
                         timestamp))
            reason = "invalid timestamp";
        else if (incidentIndex.find(id) != nullptr)
            reason = "duplicate incident ID";
        else if (junctions.find(junctionId) == nullptr)
            reason = "junction " + std::to_string(junctionId)
                     + " not found";

        if (reason.empty()) {
            Incident* incident = new Incident;

            incident->id = id;
            incident->junctionId = junctionId;
            incident->category = word[3];
            incident->severity = severity;
            incident->timestamp = timestamp;
            incident->status = "REPORTED";

            incidents.insert(id, incident);
            incidentIndex.insert(id, incident);

            std::cout << "INCIDENT REPORTED " << id << '\n';
        }

    } else if (command == "INC_FIND") {

        if (!number(word[1], 1, 999999999, id)) {
            reason = "invalid incident ID";
        } else {
            Incident* incident = incidentIndex.find(id);

            if (incident == nullptr)
                reason = "incident " + std::to_string(id)
                         + " not found";
            else
                printIncident(*incident);
        }

    } else if (command == "INC_LIST") {

        std::cout << "INCIDENTS " << incidents.size() << '\n';

        incidents.inOrder(
            [](const int&, Incident* incident) {
                printIncident(*incident);
            }
        );

    } else if (command == "INC_PREORDER") {

        std::cout << "INCIDENT PREORDER";

        if (incidents.size() == 0) {
            std::cout << " EMPTY";
        } else {
            incidents.preOrder(
                [](const int& key, Incident*) {
                    std::cout << ' ' << key;
                }
            );
        }

        std::cout << '\n';

    } else if (command == "INC_CLOSE" ||
               command == "REQUEST") {

        if (!number(word[1], 1, 999999999, id)) {
            reason = "invalid incident ID";
        } else {
            Incident* incident = incidentIndex.find(id);

            if (incident == nullptr) {
                reason = "incident " + std::to_string(id)
                         + " not found";
            } else if (command == "INC_CLOSE") {
                closeIncident(id);

                std::cout << "INCIDENT CLOSED "
                          << id << '\n';
            } else if (incident->status == "PENDING") {
                reason = "incident " + std::to_string(id)
                         + " is already PENDING";
            } else {
                incident->status = "PENDING";
                pending.insert(incident);

                std::cout << "INCIDENT PENDING "
                          << id << '\n';
            }
        }

    } else if (command == "PRIORITY_NEXT") {

        Incident* incident = pending.peek();

        if (incident == nullptr)
            std::cout << "NO PENDING INCIDENTS\n";
        else
            printIncident(*incident);

    } else if (command == "PRIORITY_SERVE") {

        Incident* incident = pending.extractMax();

        if (incident == nullptr) {
            std::cout << "NO PENDING INCIDENTS\n";
        } else {
            int servedId = incident->id;

            incidentIndex.remove(servedId);
            incidents.remove(servedId);

            std::cout << "INCIDENT SERVED "
                      << servedId << '\n';
        }

    } else if (command == "PRIORITY_COUNT") {

        std::cout << "PENDING " << pending.size() << '\n';

    } else if (command == "HASH_STATS") {

        std::cout << "HASH capacity="
                  << incidentIndex.capacity()
                  << " size="
                  << incidentIndex.size() << '\n';
    }

    if (!reason.empty()) {
        std::cout << "ERROR line " << lineNumber
                  << ": " << reason << '\n';
    }
}

#pragma once

template <typename K, typename V>
class AVL {
private:
    struct Node {
        K key;
        V value;
        int height;
        Node* left;
        Node* right;

        Node(const K& k, const V& v)
            : key(k), value(v), height(1),
              left(nullptr), right(nullptr) {}
    };

    Node* root;
    int nodeCount;

    static int nodeHeight(const Node* node) {
        return node == nullptr ? 0 : node->height;
    }

    static void updateHeight(Node* node) {
        int leftHeight = nodeHeight(node->left);
        int rightHeight = nodeHeight(node->right);

        node->height = 1 + (leftHeight > rightHeight
                            ? leftHeight : rightHeight);
    }

    static int balanceFactor(const Node* node) {
        return nodeHeight(node->left) - nodeHeight(node->right);
    }

    static Node* rotateRight(Node* y) {
        Node* x = y->left;
        Node* middle = x->right;

        x->right = y;
        y->left = middle;

        updateHeight(y);
        updateHeight(x);

        return x;
    }

    static Node* rotateLeft(Node* x) {
        Node* y = x->right;
        Node* middle = y->left;

        y->left = x;
        x->right = middle;

        updateHeight(x);
        updateHeight(y);

        return y;
    }

    Node* insertNode(Node* node, const K& key,
                     const V& value, bool& inserted) {
        if (node == nullptr) {
            inserted = true;
            ++nodeCount;
            return new Node(key, value);
        }

        if (key < node->key)
            node->left = insertNode(
                node->left, key, value, inserted);
        else if (key > node->key)
            node->right = insertNode(
                node->right, key, value, inserted);
        else
            return node;

        updateHeight(node);
        int balance = balanceFactor(node);

        if (balance > 1) {
            if (key > node->left->key)
                node->left = rotateLeft(node->left);

            return rotateRight(node);
        }

        if (balance < -1) {
            if (key < node->right->key)
                node->right = rotateRight(node->right);

            return rotateLeft(node);
        }

        return node;
    }

    template <typename Visitor>
    static void traverseInOrder(const Node* node,
                                Visitor& visit) {
        if (node == nullptr) return;

        traverseInOrder(node->left, visit);
        visit(node->key, node->value);
        traverseInOrder(node->right, visit);
    }

    template <typename Visitor>
    static void traversePreOrder(const Node* node,
                                 Visitor& visit) {
        if (node == nullptr) return;

        visit(node->key, node->value);
        traversePreOrder(node->left, visit);
        traversePreOrder(node->right, visit);
    }

    static void destroy(Node* node) {
        if (node == nullptr) return;

        destroy(node->left);
        destroy(node->right);
        delete node;
    }

public:
    AVL() : root(nullptr), nodeCount(0) {}

    ~AVL() {
        destroy(root);
    }

    AVL(const AVL&) = delete;
    AVL& operator=(const AVL&) = delete;

    bool insert(const K& key, const V& value) {
        bool inserted = false;
        root = insertNode(root, key, value, inserted);
        return inserted;
    }

    const V* find(const K& key) const {
        const Node* current = root;

        while (current != nullptr) {
            if (key == current->key)
                return &current->value;

            current = key < current->key
                ? current->left : current->right;
        }

        return nullptr;
    }

    int size() const {
        return nodeCount;
    }

    template <typename Visitor>
    void inOrder(Visitor visit) const {
        traverseInOrder(root, visit);
    }

    template <typename Visitor>
    void preOrder(Visitor visit) const {
        traversePreOrder(root, visit);
    }
};

#pragma once

template <typename K, typename V>
class BST {
private:
    struct Node {
        K key;
        V* value;
        Node* left;
        Node* right;

        Node(const K& k, V* v)
            : key(k), value(v),
              left(nullptr), right(nullptr) {}
    };

    Node* root;
    int nodeCount;

    Node* insertNode(Node* node, const K& key,
                     V* value, bool& inserted) {
        if (node == nullptr) {
            inserted = true;
            ++nodeCount;
            return new Node(key, value);
        }

        if (key < node->key)
            node->left = insertNode(
                node->left, key, value, inserted);
        else if (key > node->key)
            node->right = insertNode(
                node->right, key, value, inserted);

        return node;
    }

    static Node* detachMin(Node* node, Node*& smallest) {
        if (node->left == nullptr) {
            smallest = node;
            Node* remaining = node->right;
            node->right = nullptr;
            return remaining;
        }

        node->left = detachMin(node->left, smallest);
        return node;
    }

    Node* removeNode(Node* node, const K& key,
                     bool& removed) {
        if (node == nullptr)
            return nullptr;

        if (key < node->key) {
            node->left = removeNode(
                node->left, key, removed);
            return node;
        }

        if (key > node->key) {
            node->right = removeNode(
                node->right, key, removed);
            return node;
        }

        removed = true;
        --nodeCount;

        // Case 1 and 2: zero or one child.
        if (node->left == nullptr ||
            node->right == nullptr) {

            Node* child = node->left != nullptr
                        ? node->left : node->right;

            delete node->value;
            delete node;
            return child;
        }

        // Case 3: two children.
        Node* successor = nullptr;
        node->right = detachMin(node->right, successor);

        delete node->value;

        // Move the successor pointer, not the Incident.
        node->key = successor->key;
        node->value = successor->value;

        delete successor;
        return node;
    }

    template <typename Visitor>
    static void traverseInOrder(const Node* node,
                                Visitor& visit) {
        if (node == nullptr) return;

        traverseInOrder(node->left, visit);
        visit(node->key, node->value);
        traverseInOrder(node->right, visit);
    }

    template <typename Visitor>
    static void traversePreOrder(const Node* node,
                                 Visitor& visit) {
        if (node == nullptr) return;

        visit(node->key, node->value);
        traversePreOrder(node->left, visit);
        traversePreOrder(node->right, visit);
    }

    static void destroy(Node* node) {
        if (node == nullptr) return;

        destroy(node->left);
        destroy(node->right);

        delete node->value;
        delete node;
    }

public:
    BST() : root(nullptr), nodeCount(0) {}

    ~BST() {
        destroy(root);
    }

    BST(const BST&) = delete;
    BST& operator=(const BST&) = delete;

    // Ownership transfers only on successful insertion.
    bool insert(const K& key, V* value) {
        bool inserted = false;
        root = insertNode(root, key, value, inserted);
        return inserted;
    }

    V* find(const K& key) const {
        Node* current = root;

        while (current != nullptr) {
            if (key == current->key)
                return current->value;

            current = key < current->key
                    ? current->left : current->right;
        }

        return nullptr;
    }

    bool remove(const K& key) {
        bool removed = false;
        root = removeNode(root, key, removed);
        return removed;
    }

    int size() const {
        return nodeCount;
    }

    template <typename Visitor>
    void inOrder(Visitor visit) const {
        traverseInOrder(root, visit);
    }

    template <typename Visitor>
    void preOrder(Visitor visit) const {
        traversePreOrder(root, visit);
    }
};

#pragma once

template <typename K, typename V>
class HashMap {
private:
    struct Node {
        K key;
        V* value;
        Node* next;

        Node(const K& k, V* v, Node* n)
            : key(k), value(v), next(n) {}
    };

    Node** buckets;
    int bucketCapacity;
    int entryCount;

    static int bucketIndex(const K& key, int cap) {
        return static_cast<int>(key % cap);
    }

    static bool isPrime(int number) {
        if (number < 2) return false;

        for (int d = 2; d <= number / d; ++d) {
            if (number % d == 0) return false;
        }
        return true;
    }

    static int nextPrime(int minimum) {
        while (!isPrime(minimum))
            ++minimum;

        return minimum;
    }

    void resize() {
        int newCapacity = nextPrime(bucketCapacity * 2);
        Node** newBuckets = new Node*[newCapacity]();

        for (int i = 0; i < bucketCapacity; ++i) {
            Node* current = buckets[i];

            while (current != nullptr) {
                Node* next = current->next;
                int index = bucketIndex(
                    current->key, newCapacity);

                current->next = newBuckets[index];
                newBuckets[index] = current;
                current = next;
            }
        }

        delete[] buckets;
        buckets = newBuckets;
        bucketCapacity = newCapacity;
    }

public:
    HashMap()
        : buckets(new Node*[11]()),
          bucketCapacity(11),
          entryCount(0) {}

    ~HashMap() {
        for (int i = 0; i < bucketCapacity; ++i) {
            Node* current = buckets[i];

            while (current != nullptr) {
                Node* next = current->next;
                delete current;
                current = next;
            }
        }

        delete[] buckets;
    }

    HashMap(const HashMap&) = delete;
    HashMap& operator=(const HashMap&) = delete;

    bool insert(const K& key, V* value) {
        int index = bucketIndex(key, bucketCapacity);
        Node* current = buckets[index];

        while (current != nullptr) {
            if (current->key == key)
                return false;

            current = current->next;
        }

        buckets[index] = new Node(
            key, value, buckets[index]);

        ++entryCount;

        // Resize only when load factor exceeds 0.75.
        if (entryCount * 4 > bucketCapacity * 3)
            resize();

        return true;
    }

    V* find(const K& key) const {
        int index = bucketIndex(key, bucketCapacity);
        Node* current = buckets[index];

        while (current != nullptr) {
            if (current->key == key)
                return current->value;

            current = current->next;
        }

        return nullptr;
    }

    bool remove(const K& key) {
        int index = bucketIndex(key, bucketCapacity);
        Node* current = buckets[index];
        Node* previous = nullptr;

        while (current != nullptr) {
            if (current->key == key) {
                if (previous == nullptr)
                    buckets[index] = current->next;
                else
                    previous->next = current->next;

                // Delete the hash node, not the Incident.
                delete current;
                --entryCount;
                return true;
            }

            previous = current;
            current = current->next;
        }

        return false;
    }

    int capacity() const {
        return bucketCapacity;
    }

    int size() const {
        return entryCount;
    }
};

#pragma once
#include "Incident.h"

template <typename T>
class MaxHeap {
private:
    T* items;
    int itemCount;
    int heapCapacity;

    static bool higher(T first, T second) {
        return higherPriority(first, second);
    }

    void exchange(int a, int b) {
        T temp = items[a];
        items[a] = items[b];
        items[b] = temp;
    }

    void grow() {
        int newCapacity = heapCapacity * 2;
        T* replacement = new T[newCapacity];

        for (int i = 0; i < itemCount; ++i)
            replacement[i] = items[i];

        delete[] items;
        items = replacement;
        heapCapacity = newCapacity;
    }

    void siftUp(int index) {
        while (index > 0) {
            int parent = (index - 1) / 2;

            if (!higher(items[index], items[parent]))
                break;

            exchange(index, parent);
            index = parent;
        }
    }

    void siftDown(int index) {
        while (true) {
            int left = 2 * index + 1;
            int right = 2 * index + 2;
            int best = index;

            if (left < itemCount &&
                higher(items[left], items[best]))
                best = left;

            if (right < itemCount &&
                higher(items[right], items[best]))
                best = right;

            if (best == index)
                break;

            exchange(index, best);
            index = best;
        }
    }

public:
    MaxHeap()
        : items(new T[16]),
          itemCount(0),
          heapCapacity(16) {}

    ~MaxHeap() {
        // Incident objects belong to the BST.
        delete[] items;
    }

    MaxHeap(const MaxHeap&) = delete;
    MaxHeap& operator=(const MaxHeap&) = delete;

    void insert(T value) {
        if (itemCount == heapCapacity)
            grow();

        items[itemCount] = value;
        ++itemCount;

        siftUp(itemCount - 1);
    }

    T peek() const {
        if (itemCount == 0)
            return nullptr;

        return items[0];
    }

    T extractMax() {
        if (itemCount == 0)
            return nullptr;

        T result = items[0];
        --itemCount;

        if (itemCount > 0) {
            items[0] = items[itemCount];
            siftDown(0);
        }

        items[itemCount] = nullptr;
        return result;
    }

    bool removeById(int id) {
        int index = -1;

        // Assignment requires linear search.
        for (int i = 0; i < itemCount; ++i) {
            if (items[i]->id == id) {
                index = i;
                break;
            }
        }

        if (index == -1)
            return false;

        --itemCount;

        if (index != itemCount) {
            items[index] = items[itemCount];
            items[itemCount] = nullptr;

            if (index > 0 &&
                higher(items[index],
                       items[(index - 1) / 2])) {
                siftUp(index);
            } else {
                siftDown(index);
            }
        } else {
            items[itemCount] = nullptr;
        }

        return true;
    }

    int size() const {
        return itemCount;
    }

    int capacity() const {
        return heapCapacity;
    }
};

# TGS CityGrid - Assignment 4, Phase 1

**Student:** TY_FIND 999
**Roll number:** name, and type record. |
**Course:** Data Structures (Fall 2026)
**Language:** C++17

## Build

Run this exact build command from the project root:

```bash
g++ -std=c++17 -Wall -Icore -Iindex -Iengine engine/*.cpp -o citygrid
```

On Windows with MinGW, this produces `citygrid.exe`.

## Run

```text
citygrid <junctions file> <incidents file> <script file>
```

Example on Windows: `citygrid.exe junctions.txt incidents.txt script.txt`.
The program prints to standard output and does not read keyboard input.

## Class responsibilities

| Component | Responsibility |
|---|---|
| `core/Junction.h` | Junction ID,| `core/Incident.h` | Incident record, status, and priority comparator. |
| `index/AVL.h` | Balanced junction dictionary with insertion, search, in-order and pre-order traversals; LL/RR/LR/RL rotations. |
| `index/BST.h` | Unbalanced, ID-ordered incident registry, all three deletion cases, and memory ownership. |
| `index/HashMap.h` | Fast incident-ID lookup with separate chaining and prime-capacity resizing. |
| `index/MaxHeap.h` | Array-based emergency priority queue, peek, extract-max, remove-by-ID, and count. |
| `engine/CityGrid.h/.cpp` | File loading, parsing, validation, coordination, exact output and diagnostics. |
| `engine/CityGridCommands.cpp` | Interpret and execute all 14 commands while synchronizing structures. |
| `engine/main.cpp` | Check three command-line filenames, execute, return error/success code. |

## Hash collision strategy and resizing

I use **separate chaining** with my own singly linked nodes per bucket. The required hash function is `incidentID % capacity`. Each chain node holds a key and `Incident*`. Insertion checks duplicates. Removal unlinks and deletes only the chain node, never the Incident. The table begins with capacity **11** and immediately resizes after insertion when `size / capacity > 0.75`. The replacement capacity is the smallest prime at least twice the old capacity (11, 23, 47, 97, ...). Every live entry is rehashed. The table never shrinks. Separate chaining does not require tombstones.

## Emergency priority comparator

The highest **severity** is served first (5 beats 4). For equal severities, the **earlier timestamp** wins (smaller number). If both match, the **smaller incident ID** wins. This total ordering is used by the max-heap.

## Incident ownership and safe deletion

`INC_REPORT` or incident-file loading allocates exactly one `Incident` with `new`. The BST stores and **owns** that `Incident*`; the HashMap and MaxHeap only borrow the same pointer. For `INC_CLOSE`, the engine removes the pending heap entry, if any, then removes the hash entry, then asks the BST to delete the record and its object. `PRIORITY_SERVE` first extracts from the heap, then removes the hash entry, then deletes through the BST. This order prevents dangling references. The BST destructor deletes remaining incidents; other structure destructors free their own nodes/arrays without deleting incidents.

## Required time complexities

Let `n` be the size of the relevant structure, and `h` the actual BST height.

| Command | Required time |
|---|---|
| `CITY_ADD` | O(log n) AVL insertion |
| `CITY_FIND` | O(log n) AVL lookup |
| `CITY_LIST` | O(n) AVL in-order |
| `CITY_PREORDER` | O(n) AVL pre-order |
| `INC_REPORT` | O(h + log n) + O(1) average for BST, AVL, and hash |
| `INC_FIND` | O(1) average hash lookup |
| `INC_LIST` | O(n) BST in-order |
| `INC_PREORDER` | O(n) BST pre-order |
| `INC_CLOSE` | O(n) heap search + O(log n) heap repair + O(h) BST + O(1) average hash |
| `REQUEST` | O(1) average hash lookup + O(log n) heap insertion |
| `PRIORITY_NEXT` | O(1) heap peek |
| `PRIORITY_SERVE` | O(log n) heap extraction + O(h) BST + O(1) average hash |
| `PRIORITY_COUNT` | O(1) |
| `HASH_STATS` | O(1) |

Hash resizing takes O(n) when triggered. The BST is **not balanced**; inserting sorted incident IDs can produce a chain with `h = n`, so BST insertion, search and deletion can become O(n). The AVL guarantees O(log n) height. A collision chain may also be O(n) in a pathological hash-table case; average lookup is O(1).
